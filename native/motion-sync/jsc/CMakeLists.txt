cmake_minimum_required(VERSION 3.16)
project(cubism_motion_sync LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../src/adapter.js" MOTION_SYNC_ADAPTER)
configure_file(adapter.inc.in "${CMAKE_CURRENT_BINARY_DIR}/adapter.inc" @ONLY)
add_library(cubism_motion_sync SHARED provider.cpp ../src/decoder.c)
target_include_directories(cubism_motion_sync PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" ../vendor)
if(APPLE)
  target_link_libraries(cubism_motion_sync PRIVATE "-framework JavaScriptCore")
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(JSC REQUIRED IMPORTED_TARGET javascriptcoregtk-4.1)
  target_link_libraries(cubism_motion_sync PRIVATE PkgConfig::JSC m)
endif()
set_target_properties(cubism_motion_sync PROPERTIES C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden)
